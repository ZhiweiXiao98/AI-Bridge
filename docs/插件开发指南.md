# 插件开发教程

## 快速开始

### 1. 创建插件目录

```bash
mkdir -p plugins/panels/my_panel
cd plugins/panels/my_panel
```

### 2. 创建 plugin.json

```json
{
  "id": "my_panel",
  "name": "我的面板",
  "version": "1.0.0",
  "author": "Your Name",
  "description": "这是我的第一个面板插件",
  "icon": "🎨",
  "entry": "plugin.py",
  "class": "MyPanelPlugin",
  "default_area": "right",
  "dependencies": [],
  "min_app_version": "1.0.0"
}
```

### 3. 创建 plugin.py

```python
from app.ui.plugins.base_panel_plugin import BasePanelPlugin
from .panel import MyPanelWidget

class MyPanelPlugin(BasePanelPlugin):
    def __init__(self):
        super().__init__()
        self.panel_instance = None
    
    def on_load(self):
        print("[MyPanel] 插件已加载")
    
    def create_panel(self):
        if not self.panel_instance:
            self.panel_instance = MyPanelWidget()
        return self.panel_instance
    
    def on_panel_created(self, panel):
        print("[MyPanel] 面板已创建")
    
    def on_panel_closed(self, panel):
        print("[MyPanel] 面板已关闭")
```

### 4. 创建 panel.py

```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel
from app.ui.components.dockable_panel import DockablePanel

class MyPanelWidget(DockablePanel):
    def __init__(self, parent=None):
        super().__init__(
            panel_id="my_panel",
            title="我的面板",
            icon_name="🎨",
            parent=parent
        )
        
        # 创建内容 widget
        self.content_widget = QWidget()
        self.setWidget(self.content_widget)
        
        self.init_ui()
    
    def init_ui(self):
        layout = QVBoxLayout(self.content_widget)
        label = QLabel("欢迎使用我的面板！")
        layout.addWidget(label)
```

## 完整示例

### 创建一个计数器面板

**plugin.py**
```python
from app.ui.plugins.base_panel_plugin import BasePanelPlugin
from .panel import CounterPanelWidget

class CounterPanelPlugin(BasePanelPlugin):
    def __init__(self):
        super().__init__()
        self.panel_instance = None
        self.counter = 0
    
    def create_panel(self):
        if not self.panel_instance:
            self.panel_instance = CounterPanelWidget()
            self.panel_instance.increment_requested.connect(self.increment)
        return self.panel_instance
    
    def increment(self):
        self.counter += 1
        if self.panel_instance:
            self.panel_instance.update_count(self.counter)
```

**panel.py**
```python
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel, QPushButton
from PySide6.QtCore import Signal
from app.ui.components.dockable_panel import DockablePanel

class CounterPanelWidget(DockablePanel):
    increment_requested = Signal()
    
    def __init__(self, parent=None):
        super().__init__(
            panel_id="counter_panel",
            title="计数器",
            icon_name="🔢",
            parent=parent
        )
        
        self.content_widget = QWidget()
        self.setWidget(self.content_widget)
        self.count = 0
        
        self.init_ui()
    
    def init_ui(self):
        layout = QVBoxLayout(self.content_widget)
        
        self.label = QLabel("计数: 0")
        layout.addWidget(self.label)
        
        btn = QPushButton("增加")
        btn.clicked.connect(self.on_increment)
        layout.addWidget(btn)
    
    def on_increment(self):
        self.increment_requested.emit()
    
    def update_count(self, count):
        self.count = count
        self.label.setText(f"计数: {count}")
```

### 配置管理

```python
def get_config_schema(self):
    return {
        "auto_refresh": {
            "type": "boolean",

            "description": "自动刷新"

def load_config(self, config):
    self.config = config
    if self.panel_instance:
        self.panel_instance.apply_config(config)

def save_config(self):
    return self.config
```

### 异步操作

```python
from PySide6.QtCore import QThread, Signal

class WorkerThread(QThread):
    finished = Signal()
    
    def run(self):
        # 执行耗时操作
        result = self.do_work()
        self.finished.emit()

class MyPanelPlugin(BasePanelPlugin):
    def __init__(self):
        super().__init__()
        self.worker = None
    
    def start_work(self):
        self.worker = WorkerThread()
        self.worker.finished.connect(self.on_work_finished)
        self.worker.start()
    
    def on_work_finished(self):
        print("工作完成")
```

### 与主应用通信

```python
class MyPanelPlugin(BasePanelPlugin):
    def on_panel_created(self, panel):
        # 获取主窗口
        main_window = panel.parent()
        
        # 访问主窗口的属性
        if hasattr(main_window, 'skill_data'):
            self.skill_data = main_window.skill_data
```

### 常见错误

| 错误 | 原因 | 解决方案 |
|:---|:---|:---|
| `ModuleNotFoundError` | 导入路径错误 | 检查相对导入 |
| `AttributeError` | 属性不存在 | 检查对象初始化 |
| `TypeError` | 类型不匹配 | 检查参数类型 |

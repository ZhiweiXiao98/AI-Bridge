## 🎯 分阶段测试建设方案

### **阶段 1: 基础单元测试 (1-2 周)**

#### 目标模块
1. **`app/core/driver/__init__.py` (Facade)**
   - 测试连接初始化
   - 测试端口检测逻辑
   - 测试超时重试机制

2. **`app/core/config.py` (配置管理)**
   - 测试配置加载 (文件存在 & 不存在)
   - 测试配置保存
   - 测试默认值回退

3. **`app/ui/worker.py` (Worker 逻辑)**
   - 测试会话状态管理 (`_get_state`, `_load_states`, `_save_states`)
   - 测试信号发送（不涉及 GUI）
   - 测试哈希值比对逻辑

4. **`app/core/connection_manager.py` (连接管理)**
   - 测试消息发送/接收
   - 测试分组广播
   - 测试异常恢复

#### 预期覆盖率
- 核心业务逻辑: 60-80%
- 工具函数: 90%+
- GUI 相关: 10-20% (仅测试信号)

---

### **阶段 2: 集成测试 (2-3 周)**

#### 测试场景
1. **Chrome 连接流程**
   - 启动 → 握手 → 发送命令 → 接收响应 → 断开
   - 模拟网络超时和重连

2. **Worker 轮询循环**
   - 从 Chrome 获取消息 → 解析 → 导出文件 → 发信号

3. **Server WebSocket 端到端**
   - 客户端连接 → 鉴权 → 收发消息 → 断开
   - 多客户端并发场景

#### 使用 Mock & Fixture
```python
# 示例：Mock Chrome 连接
@pytest.fixture
def mock_chrome():
    with patch('app.core.driver.ChromeConnector') as mock:
        mock_instance = MagicMock()
        mock_instance.connect.return_value = True
        mock_instance.get_messages.return_value = [...]
        mock.return_value = mock_instance
        yield mock_instance
```

```python
import pytest
import json
import os
import tempfile
from app.core.config import ConfigManager

class TestConfigManager:
    @pytest.fixture
    def temp_config(self):
        """临时配置文件"""
        with tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False) as f:
            temp_path = f.name
        yield temp_path
        if os.path.exists(temp_path):
            os.remove(temp_path)
    
    def test_load_default_config(self):
        """测试加载默认配置"""
        config = ConfigManager.load()
        
        assert "export_code_path" in config
        assert "chrome_port" in config
        assert config["chrome_port"] == 9527
    
    def test_save_config(self):
        """测试保存配置"""
        test_config = {
            "export_code_path": "test/export",
            "chrome_port": 9999
        }
        
        ConfigManager.save(test_config)
        loaded = ConfigManager.load()
        
        assert loaded["chrome_port"] == 9999
    
    def test_config_missing_key_fallback(self):
        """测试缺失的配置键回退到默认值"""
        incomplete_config = {"chrome_port": 8888}
        ConfigManager.save(incomplete_config)
        
        loaded = ConfigManager.load()
        
        # 应该回退到默认值
        assert "export_code_path" in loaded
```

import pytest
import json

from app.ui.worker import SessionState, WorkerThread

class TestSessionState:
    def test_session_state_initialization(self):
        """测试会话状态初始化"""
        state = SessionState()
        
        assert state.max_seen_index == 0
        assert state.snapshot_index == 0
        assert state.last_hashes == []
    
    def test_session_state_update(self):
        """测试会话状态更新"""
        state = SessionState()
        state.max_seen_index = 10
        state.snapshot_index = 5
        
        assert state.max_seen_index == 10
        assert state.snapshot_index == 5

class TestWorkerStatePersistence:
    @pytest.fixture
    def temp_state_file(self):
        """临时状态文件"""
        temp_file = tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False).name
        yield temp_file
        if os.path.exists(temp_file):
            os.remove(temp_file)
    
    def test_save_and_load_states(self, temp_state_file, monkeypatch):
        """测试状态保存和加载"""

### **3. 连接管理测试** (`tests/test_connection_manager.py`)

```python
import pytest
from unittest.mock import AsyncMock, MagicMock
from app.core.connection_manager import manager

@pytest.mark.asyncio
class TestConnectionManager:
    async def test_connect_client(self):
        """测试客户端连接"""
        mock_ws = AsyncMock()
        
        await manager.connect(mock_ws, group_id="admin", client_id="client_1")
        
        assert ("admin", "client_1") in manager.active_connections
    
    async def test_send_personal_message(self):
        """测试发送个人消息"""
        mock_ws = AsyncMock()
        
        await manager.connect(mock_ws, group_id="admin", client_id="client_1")
        
        message = {"type": "test", "content": "hello"}
        await manager.send_personal_message(message, "admin", "client_1")
        
        mock_ws.send_json.assert_called()
    
    async def test_broadcast_to_group(self):
        """测试分组广播"""
        mock_ws1 = AsyncMock()
        mock_ws2 = AsyncMock()
        
        await manager.connect(mock_ws1, group_id="admin", client_id="client_1")
        await manager.connect(mock_ws2, group_id="admin", client_id="client_2")
        
        message = {"type": "broadcast", "content": "hello all"}
        await manager.broadcast_to_group(message, "admin")
        
        # 两个客户端都应该收到消息
        assert mock_ws1.send_json.called
        assert mock_ws2.send_json.called
```

## 🎓 测试最佳实践

1. **使用 Fixture 管理资源**
   ```python
   @pytest.fixture
   def mock_config():
       return {"chrome_port": 9527, "auto_export": True}
   ```

2. **使用 Mock 隔离依赖**
   ```python
   from unittest.mock import patch, MagicMock
   
   with patch('app.core.driver.ChromeConnector') as mock_driver:
       mock_driver.return_value.connect.return_value = True
   ```

3. **异步测试使用 pytest-asyncio**
   ```python
   @pytest.mark.asyncio
   async def test_async_function():
       result = await some_async_function()
       assert result is not None
   ```

4. **参数化测试减少重复代码**
   ```python
   @pytest.mark.parametrize("port,expected", [
       (9527, True),
       (65535, True),
       (-1, False),
   ])
   def test_port_validation(port, expected):
       assert is_valid_port(port) == expected
   ```
